env delete bootargs

setenv bootcmd_scsi0 'run scsi0_xen_load; run scsi0_dtb_load; run scsi0_kernel_load; run scsi0_xenpolicy_load; run scsi0_initramfs_load; bootm 0x48080000 0x84000000 0x48000000'
setenv scsi0_dtb_load 'ext2load scsi 0:1 0x48000000 xen.dtb; fdt addr 0x48000000; fdt resize; fdt mknode / boot_dev; fdt set /boot_dev device sda; fdt set /boot_dev domd_ip ${domd_ip}; fdt set /boot_dev rswitch.parallel_mode ${rswitch.parallel_mode}'
setenv scsi0_initramfs_load 'ext2load scsi 0:1 0x84000000 uInitramfs'
setenv scsi0_kernel_load 'ext2load scsi 0:1 0x7a000000 Image'
setenv scsi0_xen_load 'ext2load scsi 0:1 0x48080000 xen'
setenv scsi0_xenpolicy_load 'ext2load scsi 0:1 0x7c000000 xenpolicy'
setenv bootcmd 'run bootcmd_scsi0'
boot
